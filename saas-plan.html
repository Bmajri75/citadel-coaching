01 User Stories V1

Client (visiteur)

Client Je veux voir les coachs disponibles et leurs disciplines pour choisir celui qui me correspond.
Client Je veux remplir un formulaire de demande sans créer de compte.
Client Je veux recevoir un email de confirmation quand ma séance est validée.
Client Je veux savoir que les coachs sont diplômés d'État et vérifiés.

Coach

Coach Je veux recevoir une invitation par email pour activer mon compte.
Coach Je veux compléter mon profil (bio, photo, disciplines, zones) après activation.
Coach Je veux voir les demandes que l'admin m'attribue.
Coach Je veux accepter ou refuser une demande avec un commentaire optionnel.
Coach Je ne veux pas voir les demandes des autres coachs.

Admin (toi)

Admin Je veux créer une invitation privée pour un nouveau coach.
Admin Je veux valider le profil et la carte professionnelle d'un coach avant publication.
Admin Je veux voir toutes les demandes clients et leur statut.
Admin Je veux attribuer une demande à un coach spécifique.
Admin Je veux valider définitivement une demande acceptée par le coach.
Admin Je veux suspendre ou désactiver un coach à tout moment.
Admin Je veux un tableau de bord avec les KPI clés (demandes, coachs actifs, cartes pro à expirer).

07 Backend — SQL de création des tables

À exécuter dans l'éditeur SQL de Supabase (Table Editor → SQL Editor).

-- Profils utilisateurs (admin ou coach)
CREATE TABLE profiles (
  id          uuid REFERENCES auth.users(id) ON DELETE CASCADE PRIMARY KEY,
  role        text NOT NULL CHECK (role IN ('admin', 'coach')),
  is_active   boolean DEFAULT true,
  created_at  timestamptz DEFAULT now()
);

-- Profils métier des coachs
CREATE TABLE coaches (
  id                    uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id               uuid REFERENCES auth.users(id) ON DELETE CASCADE UNIQUE,
  prenom                text, nom text,
  bio                   text,
  photo_url             text,
  discipline_principale text,
  disciplines_sec       text[],
  zones                 text[],
  experience            text,
  diplome               text,
  num_carte_pro         text,
  expiration_carte_pro  date,
  statut_verification   text DEFAULT 'en_attente'
                        CHECK (statut_verification IN ('en_attente','verifie','expire','suspendu')),
  is_published          boolean DEFAULT false,
  created_at            timestamptz DEFAULT now(),
  updated_at            timestamptz DEFAULT now()
);

-- Invitations privées
CREATE TABLE coach_invitations (
  id          uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  token       text UNIQUE NOT NULL DEFAULT encode(gen_random_bytes(32), 'hex'),
  email       text NOT NULL,
  created_by  uuid REFERENCES auth.users(id),
  expires_at  timestamptz DEFAULT (now() + interval '48 hours'),
  used_at     timestamptz,
  is_used     boolean DEFAULT false,
  created_at  timestamptz DEFAULT now()
);

-- Demandes clients
CREATE TABLE coaching_requests (
  id                uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  prenom            text NOT NULL, nom text NOT NULL,
  email             text NOT NULL, telephone text,
  discipline        text, ville text, objectif text,
  niveau            text, disponibilites text,
  coach_preference  text, commentaire text,
  statut            text DEFAULT 'nouvelle_demande',
  consentement      boolean DEFAULT false,
  created_at        timestamptz DEFAULT now()
);

-- Attributions demande → coach
CREATE TABLE request_assignments (
  id              uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  request_id      uuid REFERENCES coaching_requests(id),
  coach_id        uuid REFERENCES auth.users(id),
  assigned_by     uuid REFERENCES auth.users(id),
  assigned_at     timestamptz DEFAULT now(),
  coach_response  text CHECK (coach_response IN ('accepted', 'refused')),
  coach_note      text,
  responded_at    timestamptz
);

-- Historique des statuts
CREATE TABLE request_status_history (
  id          uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  request_id  uuid REFERENCES coaching_requests(id),
  old_status  text, new_status text,
  changed_by  uuid REFERENCES auth.users(id),
  note        text,
  created_at  timestamptz DEFAULT now()
);

Row Level Security (qui voit quoi)

-- Activer RLS sur toutes les tables
ALTER TABLE coaches ENABLE ROW LEVEL SECURITY;
ALTER TABLE coaching_requests ENABLE ROW LEVEL SECURITY;
ALTER TABLE request_assignments ENABLE ROW LEVEL SECURITY;

-- Admin voit tout
CREATE POLICY "admin_all" ON coaching_requests
  FOR ALL USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'admin')
  );

-- Coach voit uniquement ses demandes attribuées
CREATE POLICY "coach_own_requests" ON request_assignments
  FOR SELECT USING (coach_id = auth.uid());

-- Public peut insérer une demande, pas les lire
CREATE POLICY "public_insert" ON coaching_requests
  FOR INSERT WITH CHECK (true);

-- Public voit uniquement les coachs publiés et vérifiés
CREATE POLICY "public_coaches" ON coaches
  FOR SELECT USING (is_published = true AND statut_verification = 'verifie');

Contexte d'authentification

// src/context/AuthContext.jsx
import { createContext, useContext, useEffect, useState } from 'react'
import { supabase } from '../lib/supabase'

const AuthContext = createContext({})

export function AuthProvider({ children }) {
  const [user, setUser]   = useState(null)
  const [role, setRole]   = useState(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    supabase.auth.getSession().then(async ({ data }) => {
      setUser(data.session?.user ?? null)
      if (data.session?.user) await fetchRole(data.session.user.id)
      setLoading(false)
    })
    supabase.auth.onAuthStateChange(async (_, session) => {
      setUser(session?.user ?? null)
      if (session?.user) await fetchRole(session.user.id)
      else setRole(null)
    })
  }, [])

  async function fetchRole(uid) {
    const { data } = await supabase
      .from('profiles').select('role').eq('id', uid).single()
    setRole(data?.role)
  }

  return <AuthContext.Provider value={{ user, role, loading }}>{children}</AuthContext.Provider>
}

export const useAuth = () => useContext(AuthContext)

Protection des routes par rôle

// src/components/ProtectedRoute.jsx
import { Navigate } from 'react-router-dom'
import { useAuth } from '../context/AuthContext'

export function ProtectedRoute({ children, requiredRole }) {
  const { user, role, loading } = useAuth()
  if (loading) return <div>Chargement...</div>
  if (!user) return <Navigate to="/login" />
  if (requiredRole && role !== requiredRole) return <Navigate to="/login" />
  return children
}

09 Points de sécurité critiques

🔴
Ne jamais mettre VITE_SUPABASE_SERVICE_KEY dans le frontend. La clé anon est publique par design (protégée par RLS). La clé service contourne tout — elle ne sort jamais du serveur.
PointMesure
Token invitation64 caractères hex aléatoires, expire 48h, inutilisable après usage
Mots de passeHashage automatique par Supabase Auth (bcrypt)
Accès donnéesRow Level Security — le coach ne voit que ses propres demandes
SessionsJWT signé par Supabase, HttpOnly cookie si configuré
Données publiquesSeuls les coachs is_published=true et statut=verifie sont exposés
Carte professionnelleNuméro stocké en DB, jamais affiché publiquement
Formulaire publicValidation côté client ET vérification Supabase. Insert only, pas de lecture.
